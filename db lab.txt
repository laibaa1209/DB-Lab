RETURN EMP_TBL_TYPE PIPELINED
AS
BEGIN
   FOR r IN (SELECT employee_id, first_name FROM employees
             WHERE department_id = did) LOOP
      PIPE ROW (EMP_OBJ_TYPE(r.employee_id, r.first_name));
   END LOOP;
END;

CREATE OR REPLACE FUNCTION get_details
RETURN EMP_TBL_TYPE
IS
   emp_list EMP_TBL_TYPE := EMP_TBL_TYPE();
BEGIN
   emp_list.EXTEND;

   SELECT employee_id, first_name, last_name
   INTO emp_list(1).employee_id,
        emp_list(1).first_name,
        emp_list(1).last_name
   FROM employees
   WHERE employee_id = 100;

   RETURN emp_list;
END;

CREATE OR REPLACE TRIGGER superheroes_audit
BEFORE INSERT OR DELETE OR UPDATE ON superheroes
FOR EACH ROW
BEGIN
  IF INSERTING THEN
    INSERT INTO sh_audit
    VALUES(:NEW.sh_name, NULL, SYS_CONTEXT('USERENV','CURRENT_USER'),
           TO_CHAR(SYSDATE,'DD/MON/YYYY HH24:MI:SS'), 'INSERT');

  ELSIF DELETING THEN
    INSERT INTO sh_audit
    VALUES(NULL, :OLD.sh_name, SYS_CONTEXT('USERENV','CURRENT_USER'),
           TO_CHAR(SYSDATE,'DD/MON/YYYY HH24:MI:SS'), 'DELETE');

  ELSIF UPDATING THEN
    INSERT INTO sh_audit
    VALUES(:NEW.sh_name, :OLD.sh_name, SYS_CONTEXT('USERENV','CURRENT_USER'),
           TO_CHAR(SYSDATE,'DD/MON/YYYY HH24:MI:SS'), 'UPDATE');
  END IF;
END;

-- BACKUP TABLE
CREATE TABLE superheroes_backup AS SELECT * FROM superheroes WHERE 1=2;

-- BACKUP TRIGGER
-- Keeps superheroes_backup table synchronized with superheroes
CREATE OR REPLACE TRIGGER sh_backup
BEFORE INSERT OR DELETE OR UPDATE ON superheroes
FOR EACH ROW
BEGIN
  IF INSERTING THEN
    INSERT INTO superheroes_backup (sh_name)
    VALUES(:NEW.sh_name);

  ELSIF DELETING THEN
    DELETE FROM superheroes_backup
    WHERE sh_name = :OLD.sh_name;

  ELSIF UPDATING THEN
    UPDATE superheroes_backup
    SET sh_name = :NEW.sh_name
    WHERE sh_name = :OLD.sh_name;
  END IF;
END;
/

          ORA_DICT_OBJ_TYPE,
          ORA_DICT_OBJ_NAME,
          ORA_SYSEVENT);
END;

-- CURSOR, A cursor is a temporary pointer that allows you to process query results row by row in PL/SQL.
-- cursor lifecycle
-- Declare – Define the cursor with a SELECT query.
-- Open – Execute the query and store result rows.
-- Fetch – Retrieve each row one by one.
-- Close – Release memory.
DECLARE
  CURSOR emp_cursor IS
    SELECT first_name, salary FROM employees WHERE department_id = 80;

  v_name employees.first_name%TYPE;
  v_salary employees.salary%TYPE;
BEGIN
  OPEN emp_cursor;  -- Step 1: Open the cursor this executes the SELECT query
  -- inside the cursor and stores the result set in memory
  LOOP
    FETCH emp_cursor INTO v_name, v_salary;  -- Step 2: Fetch each row
    EXIT WHEN emp_cursor%NOTFOUND;           -- Step 3: Exit when no more rows
    -- %NOTFOUND is a cursor attribute that becomes TRUE after the last row is fetched.
    DBMS_OUTPUT.PUT_LINE(v_name || ' earns ' || v_salary);
  END LOOP;
  CLOSE emp_cursor;         -- Step 4: Close the cursor
END;

---------------------------------------------------------------
-- 1. Find employees who earn more than the average salary
--    of all employees hired in the same year.
---------------------------------------------------------------
SELECT e.employee_id, e.first_name, e.salary, e.hire_date
FROM employees e
WHERE e.salary >
      ( SELECT AVG(salary)
        FROM employees
        WHERE EXTRACT(YEAR FROM hire_date) =
              EXTRACT(YEAR FROM e.hire_date) );


---------------------------------------------------------------
-- 2. List employees with NO job history, but their salary is
--    less than MIN salary of employees who HAVE job history.
---------------------------------------------------------------
SELECT e.employee_id, e.first_name, e.salary
FROM employees e
WHERE NOT EXISTS (
        SELECT 1 FROM job_history j
        WHERE j.employee_id = e.employee_id
      )
AND e.salary <
      ( SELECT MIN(e2.salary)
        FROM employees e2
        WHERE EXISTS (
            SELECT 1 FROM job_history j2
            WHERE j2.employee_id = e2.employee_id
        ) );


---------------------------------------------------------------
-- 3. Show employee name, job, salary, commission + job avg commission.
--    Only show employees whose commission > avg commission of same job.
---------------------------------------------------------------
SELECT e.first_name, e.job_id, e.salary, e.commission_pct,
       ( SELECT AVG(e2.commission_pct)
         FROM employees e2
         WHERE e2.job_id = e.job_id ) AS job_avg_commission
FROM employees e
WHERE e.commission_pct IS NOT NULL
AND e.commission_pct >
    ( SELECT AVG(e2.commission_pct)
      FROM employees e2
      WHERE e2.job_id = e.job_id );


---------------------------------------------------------------
-- 4. List employees hired in the year where the department
--    hired the MOST employees.
---------------------------------------------------------------
WITH dept_hire_counts AS (
    SELECT department_id,
           COUNT(*) AS hires,
           EXTRACT(YEAR FROM hire_date) AS hire_year
    FROM employees
    GROUP BY department_id, EXTRACT(YEAR FROM hire_date)
),
max_year AS (
    SELECT hire_year
    FROM dept_hire_counts
    GROUP BY hire_year
    ORDER BY SUM(hires) DESC
    FETCH FIRST 1 ROWS ONLY
)
SELECT e.employee_id, e.first_name, e.hire_date, d.department_name,
       ( SELECT COUNT(*)
         FROM employees e2
         WHERE EXTRACT(YEAR FROM e2.hire_date) =
               EXTRACT(YEAR FROM e.hire_date) ) AS total_hires
FROM employees e
JOIN departments d ON e.department_id = d.department_id
WHERE EXTRACT(YEAR FROM e.hire_date) = (SELECT hire_year FROM max_year);


---------------------------------------------------------------
-- 5. List departments having > 3 employees with salary > 5000.
---------------------------------------------------------------
SELECT d.department_name,
       COUNT(e.employee_id) AS employees_above_5000
FROM departments d
JOIN employees e ON d.department_id = e.department_id
WHERE e.salary > 5000
GROUP BY d.department_name
HAVING COUNT(e.employee_id) > 3;

CREATE OR REPLACE TRIGGER prevent_negative_values
BEFORE INSERT OR UPDATE ON products
FOR EACH ROW
DECLARE
BEGIN
    -----------------------------------------------------------
    -- 1. Prevent negative / zero price or stock
    -----------------------------------------------------------
    IF :NEW.price <= 0 OR :NEW.stock_quantity <= 0 THEN
        RAISE_APPLICATION_ERROR(-20001,
        'Product price or product quantity cannot be negative or zero');
    END IF;

    -----------------------------------------------------------
    -- 2. Monitor low stock levels (<5)
    -----------------------------------------------------------
    IF :NEW.stock_quantity < 5 THEN
        RAISE_APPLICATION_ERROR(-20002,
        'The product is going out of stock');
    END IF;

    -----------------------------------------------------------
    -- 3. Restrict stock reduction > 50%
    -----------------------------------------------------------
    IF UPDATING THEN
        IF :NEW.stock_quantity < (:OLD.stock_quantity * 0.5) THEN
            RAISE_APPLICATION_ERROR(-20003,
            'Stock reduction exceeds allowed limit (50%)');
        END IF;
    END IF;

    -----------------------------------------------------------
    -- 4. Automatically calculate subtotal
    -----------------------------------------------------------
    :NEW.subtotal := :NEW.price * :NEW.stock_quantity;

    -----------------------------------------------------------
    -- 5. Auto-assign audit fields
    -----------------------------------------------------------
    IF INSERTING THEN
        :NEW.created_by := USER;
    ELSIF UPDATING THEN
        :NEW.updated_by := USER;
    END IF;

    -----------------------------------------------------------
    -- 6. Update last_updated field
    -----------------------------------------------------------
    :NEW.last_updated := SYSDATE;

END;
/

/* ===========================================================
   Q3 – PL/SQL Using OBJECT TYPES (Order Billing System)
   =========================================================== */

---------------------------------------------------------------
-- 1. Create Object Type ORDER_ITEM
---------------------------------------------------------------
CREATE OR REPLACE TYPE order_item AS OBJECT(
    item_name      VARCHAR2(50),
    quantity       NUMBER,
    price          NUMBER,

    MEMBER FUNCTION total_cost RETURN NUMBER
);
/

---------------------------------------------------------------
-- 2. Create Object Type BODY with Discount Logic
---------------------------------------------------------------
CREATE OR REPLACE TYPE BODY order_item AS
    MEMBER FUNCTION total_cost RETURN NUMBER IS
        cost NUMBER;
    BEGIN
        cost := quantity * price;

        IF quantity > 5 THEN
            cost := cost * 0.95;  -- apply 5% discount
        END IF;

        RETURN cost;
    END;
END;
/

---------------------------------------------------------------
-- 3. Create Object Table
---------------------------------------------------------------
CREATE TABLE order_table OF order_item;


---------------------------------------------------------------
-- 4. PL/SQL Block to print all item totals + highest bill
---------------------------------------------------------------
DECLARE
    CURSOR c IS SELECT VALUE(o) FROM order_table o;
    v_item   order_item;
    v_cost   NUMBER;
    v_max    NUMBER := 0;
BEGIN
    DBMS_OUTPUT.PUT_LINE('Item Bill Details:');

    FOR rec IN c LOOP
        v_item := rec.VALUE;
        v_cost := v_item.total_cost;

        DBMS_OUTPUT.PUT_LINE('Item: ' || v_item.item_name ||
                             '  Final Bill: ' || v_cost);

        IF v_cost > v_max THEN
            v_max := v_cost;
        END IF;
    END LOOP;

    DBMS_OUTPUT.PUT_LINE('--------------------------------------');
    DBMS_OUTPUT.PUT_LINE('Highest Bill Amount = ' || v_max);
END;
/
